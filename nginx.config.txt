# =============================================================================
# Nginx Configuration for Streamlit with WebSocket Keepalive
# Environment : Red Hat Linux, Nginx 1.14.1
# =============================================================================
#
# CHANGES FROM PREVIOUS VERSION (nginx validation fix):
#
#   FIX 1 : Removed keepalive_requests from upstream block
#            keepalive_requests inside upstream{} requires nginx 1.15.3+.
#            On nginx 1.14.1 this causes: "unknown directive keepalive_requests"
#            and nginx -t fails. Removed entirely — not available in 1.14.1.
#
#   FIX 2 : Removed keepalive_timeout from upstream block
#            keepalive_timeout inside upstream{} also requires nginx 1.15.3+.
#            Same error as above. Removed entirely.
#            The workaround is proxy_read_timeout in each location block,
#            which controls how long nginx holds the upstream connection open
#            and is fully supported in nginx 1.14.1.
#
# WHAT IS STILL SUPPORTED IN NGINX 1.14.1 UPSTREAM BLOCK:
#   keepalive 32  — available since nginx 1.1.4, works fine on 1.14.1
#
# =============================================================================

map $http_upgrade $connection_upgrade {
    default upgrade;
    '' close;
}

upstream streamlit_backend {
    # ip_hash: ensures the same browser session always hits the same
    # Streamlit worker — required because Streamlit session state is
    # stored in-process (not shared across workers).
    ip_hash;

    server 127.0.0.1:8515 max_fails=3 fail_timeout=7200s;
    server 127.0.0.1:8516 max_fails=3 fail_timeout=7200s;
    server 127.0.0.1:8517 max_fails=3 fail_timeout=7200s;
    server 127.0.0.1:8518 max_fails=3 fail_timeout=7200s;

    # keepalive: number of idle keepalive connections nginx holds open
    # to the upstream pool. Supported since nginx 1.1.4 — safe on 1.14.1.
    # Without this, every request opens a new TCP connection to Streamlit.
    keepalive 32;

    # FIX 1 + FIX 2: keepalive_requests and keepalive_timeout are NOT
    # included here because both require nginx 1.15.3+ and cause
    # "unknown directive" errors on nginx 1.14.1.
    # The connection lifetime is instead controlled by proxy_read_timeout
    # in each location block below, which IS supported in 1.14.1.
}

server {
    listen 80;
    server_name 10.19.26.145;

    # Allow large file uploads (1 GB). Must match maxUploadSize in config.toml.
    client_max_body_size 1024M;

    # How long nginx waits for the client to send the full request body.
    # Set high for large uploads over slow internal networks.
    client_body_timeout 7200s;

    access_log /var/log/nginx/streamlit_access.log;
    error_log  /var/log/nginx/streamlit_error.log warn;

    # =========================================================================
    # BLOCK 1: DEDICATED WEBSOCKET ENDPOINT — /_stcore/stream
    # =========================================================================
    # Streamlit sends ALL real-time UI updates through this WebSocket path.
    # If this connection drops, the browser shows a blank screen or spinner
    # even though the Python process is still running on the server.
    # =========================================================================
    location /_stcore/stream {
        proxy_pass http://streamlit_backend;

        # Required for WebSocket: tell upstream to switch protocols.
        # Without HTTP/1.1, WebSocket upgrade is not possible.
        proxy_http_version 1.1;
        proxy_set_header Upgrade    $http_upgrade;
        proxy_set_header Connection $connection_upgrade;

        # Standard reverse proxy identification headers.
        proxy_set_header Host              $host;
        proxy_set_header X-Real-IP         $remote_addr;
        proxy_set_header X-Forwarded-For   $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        # proxy_socket_keepalive is NOT included — requires nginx 1.15.6+.
        # Workaround: X-Accel-Buffering tells Streamlit not to buffer responses,
        # and tcp_nodelay sends WebSocket frames immediately (Nagle algo off).
        proxy_set_header X-Accel-Buffering no;
        tcp_nodelay on;

        # Timeouts set to 7200s (2 hours) for long-running mapping operations.
        # proxy_read_timeout is the most critical — it controls how long nginx
        # waits between WebSocket data frames before closing the connection.
        # This also acts as the upstream keepalive lifetime workaround for
        # the missing keepalive_timeout in the upstream block on 1.14.1.
        proxy_connect_timeout 7200s;
        proxy_send_timeout    7200s;
        proxy_read_timeout    7200s;
        send_timeout          7200s;

        # Disable buffering — required for WebSocket streaming.
        proxy_request_buffering off;
        proxy_buffering         off;

        # Do NOT try another backend if this one fails mid-WebSocket.
        # Session state is lost the moment the connection drops.
        proxy_next_upstream off;

        # Increased buffer sizes to handle large Streamlit DataFrame payloads.
        # Default (4k/8k) causes "upstream sent too big header" on large results.
        proxy_buffer_size       256k;
        proxy_buffers           8 256k;
        proxy_busy_buffers_size 512k;

        # Disable gzip — compression is incompatible with WebSocket framing
        # and causes silent data corruption or connection resets.
        gzip off;
    }

    # =========================================================================
    # BLOCK 2: ALL OTHER HTTP REQUESTS
    # =========================================================================
    # Handles page loads, static assets, API calls, and file uploads.
    # =========================================================================
    location / {
        proxy_pass http://streamlit_backend;

        # HTTP/1.1 required to use the upstream keepalive pool.
        proxy_http_version 1.1;

        proxy_set_header Upgrade    $http_upgrade;

        # Empty Connection header for HTTP requests so nginx uses the
        # upstream keepalive pool instead of closing after each request.
        proxy_set_header Connection "";

        proxy_set_header Host              $host;
        proxy_set_header X-Real-IP         $remote_addr;
        proxy_set_header X-Forwarded-For   $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Accel-Buffering no;

        tcp_nodelay on;

        proxy_connect_timeout 7200s;
        proxy_send_timeout    7200s;
        proxy_read_timeout    7200s;
        send_timeout          7200s;
        keepalive_timeout     7200s;

        proxy_request_buffering off;
        proxy_buffering         off;

        proxy_next_upstream error timeout invalid_header http_500 http_502 http_503;
    }

    # =========================================================================
    # BLOCK 3: HEALTH CHECK ENDPOINT
    # =========================================================================
    location /_health {
        access_log off;
        return 200 "healthy\n";
        add_header Content-Type text/plain;
    }
}

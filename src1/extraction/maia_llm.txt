"""
Maia LLM Client for PDF Extraction
Provides integration with Maia API models (gpt-5.1, etc.)
"""

import base64
import datetime
import httpx
import json
import logging
import os
import ssl
import requests
from typing import Optional, Dict, Any

# Maia API Configuration
MAIA_API_ENDPOINT = "https://api.ai.uat.intranet.groupebpce.fr"
MAIA_API_VERSION = "v2"
DEFAULT_MAX_TOKEN = 4096
DEFAULT_TEMPERATURE = 0.0
DEFAULT_CONTENT_TYPE = "application/json"
DEFAULT_OUTPUT_FORMAT = "TEXT"
DEFAULT_PROMPT_SYSTEM = "you're a useful assistant"
DEFAULT_REQUEST_TIMEOUT = 100
DEFAULT_TOKEN_REQUEST_TIMEOUT = 10

ENDPOINTS = {
    "token": "/oauth2/token",
    "tokens": f"/generativeAI/{MAIA_API_VERSION}/texts/tokens",
    "models": f"/generativeAI/{MAIA_API_VERSION}/models?pageSize=100",
    "texts": f"/generativeAI/{MAIA_API_VERSION}/texts"
}


class MaiaAPIClient:
    """
    Handles authentication and communication with the Maia generative AI API.
    """
    
    def __init__(self, credentials: str, proxies: Optional[Dict[str, str]] = None, verify=False, use_proxy: bool = True):
        """
        Initialize the Maia API client.
        
        Args:
            credentials (str): API credentials in format "client_id:client_secret"
            proxies (Optional[Dict[str, str]]): Proxy settings
            verify (bool): SSL verification flag
            use_proxy (bool): Whether to use proxy (False to bypass proxy completely)
        """
        self.encoded_credentials = base64.b64encode(credentials.encode('utf8')).decode('utf8')
        self.token = None
        self.token_expiry = None
        
        # Convert proxies dict to httpx format
        proxy_url = None
        
        # Only use proxy if explicitly enabled
        if use_proxy:
            if proxies:
                # Use https proxy if available, otherwise http
                proxy_url = proxies.get("https://") or proxies.get("http://")
            elif os.getenv("https_proxy"):
                proxy_url = os.getenv("https_proxy")
            elif os.getenv("http_proxy"):
                proxy_url = os.getenv("http_proxy")
        
        # Disable SSL warnings when verify=False
        if not verify:
            try:
                import urllib3
                urllib3.disable_warnings(urllib3.exceptions.InsecureRequestWarning)
            except:
                pass
        
        # CRITICAL FIX: For .intranet domains, use requests library instead of httpx
        # httpx has issues with system-level proxies that cannot be bypassed
        # requests library with session.trust_env=False works better
        if ".intranet" in MAIA_API_ENDPOINT and not use_proxy:
            logging.info("Maia API using requests library (bypassing httpx proxy issues)")
            # Use requests library with explicit no-proxy session
            self.use_requests = True
            self.session = requests.Session()
            self.session.trust_env = False  # Don't read proxy from environment
            self.session.verify = False  # Disable SSL verification
            self.session.proxies = {}  # Explicitly no proxy
            self.timeout = 30.0
            self.client = None  # Not using httpx
        elif proxy_url:
            logging.info(f"Maia API using proxy: {proxy_url}")
            self.use_requests = False
            self.client = httpx.Client(verify=False, proxy=proxy_url, timeout=30.0)
        else:
            logging.info("Maia API using direct connection (no proxy)")
            self.use_requests = False
            # Use mounts={{}} to completely bypass proxy (like we do for Gemma)
            self.client = httpx.Client(verify=False, mounts={}, timeout=30.0)
        
    def _request(self, method: str, endpoint_key: str, headers: Optional[Dict[str, str]] = None, 
                 json_data=None, data: Optional[Dict] = None, timeout: int = DEFAULT_REQUEST_TIMEOUT):
        """Send an HTTP request to the API."""
        url = f"{MAIA_API_ENDPOINT}{ENDPOINTS[endpoint_key]}"
        response = None
        try:
            # Log request details for debugging
            logging.debug(f"Maia API Request: {method} {url}")
            if json_data:
                logging.debug(f"Request payload: {json.dumps(json_data, indent=2)[:500]}")
            
            # CRITICAL: Use requests library for .intranet domains to bypass proxy
            if self.use_requests:
                logging.debug("Using requests library (trust_env=False, proxies={})")
                response = self.session.request(
                    method, url, 
                    headers=headers, 
                    json=json_data, 
                    data=data, 
                    timeout=timeout or self.timeout
                )
            else:
                response = self.client.request(method, url, headers=headers, json=json_data, 
                                              data=data, timeout=timeout)
            
            response.raise_for_status()
            return response.json()
        except (httpx.HTTPStatusError, requests.HTTPError) as e:
            # Try to parse JSON error, but handle cases where response is not JSON (e.g., proxy errors)
            try:
                error_detail = e.response.json()
                logging.error(f"Maia API request failed: {e.response.status_code}, {error_detail}")
                logging.error(f"Request payload was: {json.dumps(json_data, indent=2) if json_data else 'None'}")
            except:
                logging.error(f"Maia API request failed: {e.response.status_code}, Response: {e.response.text[:500]}")
                logging.error(f"Request payload was: {json.dumps(json_data, indent=2) if json_data else 'None'}")
            raise Exception(f"Maia API error: {e.response.status_code} - Check request payload and credentials")
        except (httpx.ConnectError, requests.ConnectionError) as e:
            logging.error(f"Maia API connection error: {str(e)}")
            raise Exception(f"Cannot connect to Maia API. Check proxy settings: {str(e)}")
        except Exception as e:
            logging.error(f"Maia API error: {str(e)}")
            raise
        return None
    
    def _get_bearer(self):
        """Retrieve and cache the bearer token for authentication."""
        if self.token and self.token_expiry > datetime.datetime.now():
            logging.info("Reusing cached Maia token")
            return self.token
        
        headers = {
            "Authorization": f"Basic {self.encoded_credentials}",
            "Content-Type": "application/x-www-form-urlencoded"
        }
        response = self._request("POST", "token", headers=headers, 
                                data={'grant_type': 'client_credentials'}, 
                                timeout=DEFAULT_TOKEN_REQUEST_TIMEOUT)
        
        if response:
            self.token = response['access_token']
            self.token_expiry = datetime.datetime.now() + datetime.timedelta(seconds=response['expires_in'])
            logging.info("Maia access granted, token cached")
            return self.token
        return None
    
    def _get_headers(self, content_type=DEFAULT_CONTENT_TYPE):
        """Construct headers for API requests."""
        bearer_token = self.token if self.token and datetime.datetime.now() <= self.token_expiry else self._get_bearer()
        
        if bearer_token:
            return {
                "Authorization": f"Bearer {bearer_token}",
                "Content-Type": content_type
            }
        return None
    
    def _json_payload(self, **kwargs):
        """Construct a JSON payload for API chat requests."""
        output_format = kwargs.get("outputFormat", DEFAULT_OUTPUT_FORMAT).upper()
        prompt = kwargs.get("prompt", "")
        prompt_system = kwargs.get("prompt_system", DEFAULT_PROMPT_SYSTEM)
        model_id = kwargs.get("modelId", "gpt-4o-mini-2024-07-18")
        temperature = kwargs.get("temperature", DEFAULT_TEMPERATURE)
        output_max_tokens = kwargs.get("outputMaxTokens", DEFAULT_MAX_TOKEN)
        
        # IMPORTANT: Don't modify prompt - it already requests JSON format
        # if output_format.upper() == "JSON":
        #     prompt += " return in a JSON format."
        
        # Use the original MAIA starter pack format: list of content dicts with "text" key
        contents = kwargs.get("contents", [{"text": prompt}])
        
        payload = {
            "modelId": model_id,
            "input": {
                "systemInstructions": [{"text": prompt_system}],
                "contents": contents
            },
            "generationConfig": {
                "outputFormat": output_format,
                "temperature": temperature,
                "outputMaxTokens": output_max_tokens,
            }
        }
        
        logging.debug(f"Constructed MAIA payload: {json.dumps(payload, indent=2)[:500]}")
        return payload
    
    def chat(self, **kwargs):
        """
        Send a chat request to the model and retrieve the response.
        
        Args:
            **kwargs: Arbitrary keyword arguments to customize the chat request.
                prompt (str): The prompt text
                modelId (str): The model identifier
                prompt_system (str): System instructions
                temperature (float): Sampling temperature
                outputMaxTokens (int): Maximum output tokens
                outputFormat (str): Output format (TEXT or JSON)
        
        Returns:
            dict or None: Response from the API if successful, otherwise None
        """
        payload_json = self._json_payload(**kwargs)
        headers = self._get_headers()
        
        if headers:
            response = self._request("POST", "texts", headers=headers, 
                                    json_data=payload_json, timeout=100)
            return response
        return None


class MaiaOpenAIAdapter:
    """
    Adapter to make Maia API compatible with OpenAI client interface.
    This allows seamless switching between Gemma and Maia models.
    """
    
    def __init__(self, api_client: MaiaAPIClient, model: str):
        self.api_client = api_client
        self.model = model
        self.chat = self.ChatCompletions(self)
    
    class ChatCompletions:
        def __init__(self, parent):
            self.parent = parent
            self.completions = self
        
        def create(self, model: str, messages: list, temperature: float = 0, 
                  response_format: Optional[Dict] = None, stream: bool = False, **kwargs):
            """
            Create a chat completion (OpenAI-compatible interface).
            
            Args:
                model (str): Model identifier
                messages (list): List of message dicts with 'role' and 'content'
                temperature (float): Sampling temperature
                response_format (dict): Response format specification (IGNORED - not supported by MAIA)
                stream (bool): Whether to stream the response (Maia doesn't support streaming, will return full response)
                **kwargs: Additional arguments
            
            Returns:
                MaiaResponse or generator: Response object compatible with OpenAI format
            """
            # Extract system and user messages
            system_msg = ""
            user_msg = ""
            
            for msg in messages:
                if msg.get("role") == "system":
                    system_msg = msg.get("content", "")
                elif msg.get("role") == "user":
                    user_msg = msg.get("content", "")
            
            # IMPORTANT: Don't use outputFormat="JSON" - it causes 400 errors
            # The prompt itself should request JSON format
            # output_format = "JSON" if response_format and response_format.get("type") == "json_object" else "TEXT"
            output_format = "TEXT"  # Always use TEXT format
            
            logging.debug(f"MAIA API call - Model: {model}, Temperature: {temperature}, Format: {output_format}")
            logging.debug(f"System message: {system_msg[:100]}...")
            logging.debug(f"User message: {user_msg[:200]}...")
            
            # Call Maia API
            response = self.parent.api_client.chat(
                prompt=user_msg,
                prompt_system=system_msg or DEFAULT_PROMPT_SYSTEM,
                modelId=model,
                temperature=temperature,
                outputFormat=output_format,
                outputMaxTokens=kwargs.get("max_tokens", DEFAULT_MAX_TOKEN)
            )
            
            if not response:
                raise Exception("Maia API returned no response")
            
            # If streaming is requested, simulate it by yielding the full response as chunks
            if stream:
                return self._simulate_stream(response, kwargs.get("stream_options", {}))
            
            # Convert to OpenAI-compatible format
            return MaiaResponse(response)
        
        def _simulate_stream(self, maia_response: Dict, stream_options: Dict):
            """
            Simulate streaming by yielding the full response.
            Maia API doesn't support true streaming, so we return the complete response.
            """
            candidates = maia_response.get("candidates", [])
            content = candidates[0].get("text", "") if candidates else ""
            
            # Yield the content as a single chunk
            yield MaiaStreamChunk(content, is_final=False)
            
            # If stream_options includes usage, yield a final chunk with usage
            if stream_options.get("include_usage"):
                usage_metadata = maia_response.get("usageMetadata", {})
                yield MaiaStreamChunk("", is_final=True, usage_metadata=usage_metadata)
    
    class Usage:
        def __init__(self, usage_metadata: Dict):
            self.prompt_tokens = usage_metadata.get("inputTokens", 0)
            self.completion_tokens = usage_metadata.get("outputTokens", 0)
            self.total_tokens = usage_metadata.get("totalTokens", 0)


class MaiaResponse:
    """OpenAI-compatible response object for Maia API responses."""
    
    def __init__(self, maia_response: Dict):
        self.maia_response = maia_response
        self.choices = [self.Choice(maia_response)]
        
        usage_metadata = maia_response.get("usageMetadata", {})
        self.usage = MaiaOpenAIAdapter.Usage(usage_metadata)
    
    class Choice:
        def __init__(self, maia_response: Dict):
            candidates = maia_response.get("candidates", [])
            content = candidates[0].get("text", "") if candidates else ""
            self.message = self.Message(content)
        
        class Message:
            def __init__(self, content: str):
                self.content = content


class MaiaStreamChunk:
    """OpenAI-compatible streaming chunk for Maia API responses."""
    
    def __init__(self, content: str, is_final: bool = False, usage_metadata: Optional[Dict] = None):
        self.choices = [self.Choice(content)] if not is_final else []
        self.usage = MaiaOpenAIAdapter.Usage(usage_metadata) if usage_metadata else None
    
    class Choice:
        def __init__(self, content: str):
            self.delta = self.Delta(content)
        
        class Delta:
            def __init__(self, content: str):
                self.content = content


def get_maia_client(credentials: str, model: str, proxies: Optional[Dict] = None, use_proxy: bool = True) -> MaiaOpenAIAdapter:
    """
    Get a Maia client with OpenAI-compatible interface.
    
    Args:
        credentials (str): Maia API credentials
        model (str): Model identifier
        proxies (Optional[Dict]): Proxy configuration
        use_proxy (bool): Whether to use proxy (False to bypass completely)
    
    Returns:
        MaiaOpenAIAdapter: Client compatible with OpenAI interface
    """
    api_client = MaiaAPIClient(credentials, proxies=proxies, verify=False, use_proxy=use_proxy)
    return MaiaOpenAIAdapter(api_client, model)

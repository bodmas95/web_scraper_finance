def _curl_get_json(url: str, headers: dict, timeout: int) -> Any:
    """Fallback: fetch JSON via curl subprocess with NTLM proxy auth."""
    import subprocess

    proxy_url = f"http://{settings.PROXY_CORPORATE_HOST}:{settings.PROXY_CORPORATE_PORT}"
    cmd = [
        "curl", "--silent", "--show-error", "--location", "--ssl-no-revoke",
        "--proxy", proxy_url,
        "--proxy-ntlm",
        "-U", f"{settings.PROXY_CORPORATE_USERNAME}:{settings.PROXY_CORPORATE_PASSWORD}",
        "--max-time", str(timeout),
    ]
    for k, v in headers.items():
        cmd.extend(["-H", f"{k}: {v}"])
    cmd.append(url)

    res = subprocess.run(cmd, capture_output=True, timeout=timeout + 30)
    if res.returncode != 0:
        raise RuntimeError(
            f"curl failed ({res.returncode}) for {url}: "
            f"{res.stderr.decode(errors='replace')}"
        )
    return json.loads(res.stdout)


def _pycurl_get_json(url: str, headers: dict, timeout: int) -> Any:
    """Fetch JSON via pycurl with NTLM proxy auth."""
    import pycurl

    buf = BytesIO()
    c = pycurl.Curl()
    try:
        c.setopt(pycurl.URL, url)
        c.setopt(pycurl.PROXY, settings.PROXY_CORPORATE_HOST)
        c.setopt(pycurl.PROXYPORT, settings.PROXY_CORPORATE_PORT)
        c.setopt(pycurl.PROXYUSERNAME, settings.PROXY_CORPORATE_USERNAME)
        c.setopt(pycurl.PROXYPASSWORD, settings.PROXY_CORPORATE_PASSWORD)
        c.setopt(pycurl.PROXYAUTH, pycurl.HTTPAUTH_NTLM)
        c.setopt(pycurl.SSL_VERIFYPEER, 0)
        c.setopt(pycurl.SSL_VERIFYHOST, 0)
        c.setopt(pycurl.TIMEOUT, timeout)
        c.setopt(pycurl.WRITEDATA, buf)
        c.setopt(pycurl.HTTPHEADER, [f"{k}: {v}" for k, v in headers.items()])
        c.perform()
        status = c.getinfo(pycurl.RESPONSE_CODE)
    finally:
        c.close()

    if status >= 400:
        raise RuntimeError(f"HTTP {status} for {url}")
    return json.loads(buf.getvalue())


def _http_get_json(url: str, params: dict = None, timeout: int = 60) -> Any:
    headers = {
        "Accept": "application/json,*/*",
        "User-Agent": "XBRL-Research/1.0 research@natixis.com",
    }

    if settings.PROXY_MODE == "system":
        if params:
            url = f"{url}?{urlencode(params, doseq=True)}"

        try:
            return _pycurl_get_json(url, headers, timeout)
        except ImportError:
            logger.warning("pycurl not installed, falling back to curl subprocess")
            return _curl_get_json(url, headers, timeout)

    else:
        proxies = None
        if settings.PROXY_MODE == "server":
            p = f"http://{settings.PROXY_SERVER_HOST}:{settings.PROXY_SERVER_PORT}"
            proxies = {"http": p, "https": p}

        resp = req_lib.get(
            url, params=params, headers=headers,
            timeout=timeout, proxies=proxies, verify=False,
        )
        resp.raise_for_status()
        return resp.json()


def _parse_xbrl_value(val) -> float | None:
    if val is None:
        return None
    if isinstance(val, (int, float)):
        return float(val)
    s = str(val).replace(",", "").replace(" ", "").strip()
    if not s or s in ("", "nan", "None"):
        return None
    try:
        return float(s)
    except ValueError:
        return None


def _parse_period(period: str) -> tuple[str, str, str, str]:
    """Parse XBRL period → (type, start, end, fy_year). Matches web_scraper logic."""
    if not period:
        return "unknown", "", "", ""
    if "/" in period:
        start, end = period.split("/", 1)
        end_date = end.strip()[:10]
        if end_date[5:] == "01-01":
            fy_year = str(int(end_date[:4]) - 1)
        else:
            fy_year = end_date[:4]
        return "duration", start.strip(), end.strip(), fy_year
    else:
        date = period.strip()[:10]
        if len(date) >= 10 and date[5:10] == "01-01":
            fy_year = str(int(date[:4]) - 1)
        else:
            fy_year = date[:4]
        return "instant", "", period.strip(), fy_year


def _fetch_filings(lei: str) -> list[dict]:
    """Fetch filing list from filings.xbrl.org OIM API."""
    api_base = settings.XBRL_API_BASE
    data = _http_get_json(
        f"{api_base}/api/filings",
        params={"filter[entity.identifier]": lei, "page[size]": 50},
        timeout=30,
    )
    filings_raw = data.get("data", [])
    filings = []
    for f in filings_raw:
        attrs = dict(f.get("attributes", {}))
        attrs["_id"] = f.get("id", "")
        filings.append(attrs)
    filings.sort(key=lambda x: x.get("period_end", ""), reverse=True)
    return filings


def _fetch_xbrl_facts(filing: dict) -> list[dict]:
    """Download OIM xBRL-JSON for a filing and extract all facts."""
    json_url = filing.get("json_url", "")
    if not json_url:
        return []

    api_base = settings.XBRL_API_BASE
    full_url = api_base + json_url if not json_url.startswith("http") else json_url
    data = _http_get_json(full_url, timeout=120)

    facts_raw = data.get("facts", {})
    result = []

    for _fact_id, fact in facts_raw.items():
        dims = fact.get("dimensions", {})
        concept = dims.get("concept", "")
        period = dims.get("period", "")
        unit = dims.get("unit", "")
        raw_val = fact.get("value", "")
        decimals = fact.get("decimals", "")

        if not concept:
            continue

        p_type, p_start, p_end, fy_year = _parse_period(period)
        numeric = _parse_xbrl_value(raw_val)
        concept_short = concept.split(":")[-1] if ":" in concept else concept

        result.append({
            "concept_full": concept,
            "concept_short": concept_short,
            "period_type": p_type,
            "period_start": p_start,
            "period_end": p_end,
            "fy_year": fy_year,
            "unit": unit,
            "value": numeric,
            "decimals": decimals,
        })

    return result


def _extract_currency_and_scale(facts: list[dict]) -> tuple[str, str | None, int]:
    currency = "EUR"
    for fact in facts:
        if fact.get("value") is not None:
            unit = fact.get("unit", "")
            if unit:
                currency = unit.split(":")[-1].upper() if ":" in unit else unit.upper()
            break

    decimals_values = []
    for fact in facts:
        if fact.get("value") is not None and fact.get("decimals"):
            try:
                decimals_values.append(int(fact["decimals"]))
            except (ValueError, TypeError):
                pass

    if decimals_values:
        most_common = Counter(decimals_values).most_common(1)[0][0]
        if most_common <= -6:
            return currency, "Millions", 1_000_000
        elif most_common <= -3:
            return currency, "Thousands", 1_000

    values = [abs(f["value"]) for f in facts if f.get("value") and f["value"] != 0]
    if values:
        median = sorted(values)[len(values) // 2]
        if median >= 1_000_000:
            return currency, "Millions", 1_000_000
        elif median >= 1_000:
            return currency, "Thousands", 1_000

    return currency, None, 1


def _build_consolidated(all_facts: list[dict], filing_years: list[str]) -> dict:
    """Build consolidated multi-year statements from all XBRL facts.
    
    Args:
        all_facts: List of XBRL facts from all filings
        filing_years: List of fiscal years from the filing period_end dates
    """
    # Use the filing years as the authoritative source
    keep_years = set(filing_years[:3])  # Keep up to 3 years
    
    logger.info("Filing years to keep: %s", sorted(keep_years, reverse=True))
    
    # Also log what years are in the facts for debugging
    duration_years = sorted(
        {f["fy_year"] for f in all_facts
         if f.get("fy_year") and f.get("period_type") == "duration"},
        reverse=True,
    )
    logger.info("Duration years found in facts: %s", duration_years)

    currency, unit_label, divisor = _extract_currency_and_scale(all_facts)

    grouped: dict[tuple, dict] = {}
    concept_full_map: dict[str, str] = {}

    for fact in all_facts:
        fy = fact.get("fy_year", "")
        if fy not in keep_years:
            continue

        cs = fact["concept_short"]
        cf = fact["concept_full"]
        stmt_type = STATEMENT_MAP.get(cs)
        if not stmt_type:
            continue

        val = fact.get("value")
        if val is None:
            continue

        key = (cs, stmt_type)
        if key not in grouped:
            grouped[key] = {}
            concept_full_map[cs] = cf

        if fy not in grouped[key]:
            grouped[key][fy] = val

    years = sorted(keep_years, reverse=True)

    result = {}
    for stmt_type in ("income_statement", "balance_sheet", "cash_flow"):
        rows = []
        for (cs, st), year_vals in grouped.items():
            if st != stmt_type:
                continue

            label = CONCEPT_LABELS.get(cs, cs)
            row: dict[str, Any] = {
                "label": label,
                "concept": concept_full_map.get(cs, cs),
                "standard_concept": cs,
            }

            for yr in years:
                if yr in year_vals:
                    val = year_vals[yr]
                    if divisor > 1:
                        val = val / divisor
                    row[yr] = round(val, 1)

            rows.append(row)

        result[stmt_type] = {
            "rows": rows,
            "year_headers": years,
            "unit_scale": unit_label,
            "currency": currency,
        }

    return result


def _fetch_sync(lei: str) -> dict:
    logger.info("Fetching XBRL filings for LEI=%s", lei)

    filings = _fetch_filings(lei)
    if not filings:
        raise ValueError(f"No XBRL filings found for LEI {lei}")

    logger.info("Found %d filings for LEI=%s", len(filings), lei)
    
    # Log first 5 filings to debug year issues
    for i, f in enumerate(filings[:5]):
        logger.info("  Filing %d: period_end=%s, period_start=%s", 
                   i+1, f.get("period_end", ""), f.get("period_start", ""))

    # Extract actual filing years from period_end dates
    filing_years = []
    for f in filings[:3]:  # Only use the 3 most recent filings
        pe = f.get("period_end", "")
        if pe and len(pe) >= 10:
            # Parse the fiscal year from period_end
            end_date = pe[:10]
            if end_date[5:10] == "01-01":
                # If period ends on Jan 1, fiscal year is previous year
                fy = str(int(end_date[:4]) - 1)
            else:
                fy = end_date[:4]
            filing_years.append(fy)
    
    logger.info("Filing years (from period_end): %s", filing_years)

    all_facts = []
    for filing in filings[:3]:
        logger.info("Fetching facts: period_end=%s", filing.get("period_end"))
        facts = _fetch_xbrl_facts(filing)
        all_facts.extend(facts)
        logger.info("Got %d facts", len(facts))

    if not all_facts:
        raise ValueError(f"No XBRL facts found for LEI {lei}")

    result = _build_consolidated(all_facts, filing_years)
    result["filing_years"] = filing_years
    return result


async def fetch_xbrl_financials(lei: str, year: int = None) -> dict:
    result = await asyncio.to_thread(_fetch_sync, lei)

    for key in ("income_statement", "balance_sheet", "cash_flow"):
        if result.get(key) is None:
            result[key] = {
                "rows": [],
                "year_headers": [],
                "unit_scale": None,
                "currency": None,
            }

    logger.info(
        "XBRL extraction complete: IS=%d rows, BS=%d rows, CF=%d rows",
        len(result["income_statement"]["rows"]),
        len(result["balance_sheet"]["rows"]),
        len(result["cash_flow"]["rows"]),
    )
    return result

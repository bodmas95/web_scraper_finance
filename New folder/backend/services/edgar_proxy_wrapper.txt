"""
EDGAR API Proxy Wrapper
Patches the edgar-tool library to work with corporate NTLM proxy
"""
import subprocess
import json
import os
import logging

logger = logging.getLogger(__name__)


def setup_edgar_with_proxy(config_settings):
    """
    Setup EDGAR library to work with corporate proxy using NTLM authentication.
    Reads proxy settings from config and patches edgar library functions.
    
    Args:
        config_settings: Settings object with proxy configuration
    """
    proxy_mode = config_settings.PROXY_MODE
    
    if proxy_mode == 'none':
        logger.info("📡 EDGAR: Direct connection (no proxy)")
        from edgar import set_identity
        set_identity(config_settings.EDGAR_IDENTITY)
        return
    
    elif proxy_mode == 'server':
        # IP-based proxy (no authentication) - Patch httpx to use proxy
        proxy_host = config_settings.PROXY_SERVER_HOST
        proxy_port = config_settings.PROXY_SERVER_PORT
        PROXY = f"http://{proxy_host}:{proxy_port}"
        logger.info("EDGAR: Using server proxy %s (no auth)", PROXY)

        # Patch httpx.Client to always use proxy (scoped to httpx only,
        # no global HTTP_PROXY env vars that would leak into requests/XBRL)
        import httpx

        _original_client_init = httpx.Client.__init__

        def patched_client_init(self, *args, **kwargs):
            if 'proxy' not in kwargs and 'proxies' not in kwargs:
                kwargs['proxy'] = PROXY
            if 'verify' not in kwargs:
                kwargs['verify'] = False
            _original_client_init(self, *args, **kwargs)

        httpx.Client.__init__ = patched_client_init

        try:
            import edgar.httprequests as hr
            if hasattr(hr, 'client'):
                hr.client = httpx.Client(proxy=PROXY, verify=False)
        except Exception as e:
            logger.warning("Error patching edgar.httprequests: %s", e)

        from edgar import set_identity
        set_identity(config_settings.EDGAR_IDENTITY)

        logger.info("EDGAR configured with server proxy (httpx patched)")
        return
        
    elif proxy_mode == 'system':
        # NTLM proxy (with authentication)
        proxy_host = config_settings.PROXY_CORPORATE_HOST
        proxy_port = config_settings.PROXY_CORPORATE_PORT
        USER = config_settings.PROXY_CORPORATE_USERNAME
        PASSWORD = config_settings.PROXY_CORPORATE_PASSWORD
        PROXY = f"http://{proxy_host}:{proxy_port}"
        logger.info(f"📡 EDGAR: Using corporate proxy {PROXY} with NTLM auth")
    
    else:
        raise ValueError(f"Invalid PROXY_MODE value: {proxy_mode}")
    
    # Get user agent
    UA = config_settings.EDGAR_IDENTITY
    
    # Define curl-based fetch function
    def curl_fetch(url: str, as_text=False):
        """Fetch URL using curl with NTLM proxy authentication"""
        cmd = [
            "curl",
            "--silent",
            "--show-error",
            "--location",
            "--ssl-no-revoke",
        ]
        
        # Add proxy authentication if using NTLM
        if proxy_mode == 'system' and USER and PASSWORD:
            cmd.extend([
                "--proxy-ntlm",
                "-U", f"{USER}:{PASSWORD}",
            ])
        
        # Add proxy
        cmd.extend([
            "--proxy", PROXY,
            "-A", UA,
            url,
        ])
        
        res = subprocess.run(cmd, capture_output=True)
        if res.returncode != 0:
            raise RuntimeError(
                f"Curl failed for {url}\n"
                f"returncode={res.returncode}\n"
                f"stderr={res.stderr.decode(errors='replace')}"
            )
        return res.stdout.decode("utf-8", errors="replace") if as_text else res.stdout
    
    # Define FakeResponse class to mimic httpx.Response
    class FakeResponse:
        def __init__(self, url, content: bytes, status_code: int = 200):
            self.url = url
            self._content = content
            self.status_code = status_code
            self.headers = {"content-type": "text/html"}

        @property
        def content(self):
            return self._content

        @property
        def text(self):
            return self._content.decode("utf-8", errors="replace")

        def json(self):
            return json.loads(self.text)

        def read(self):
            return self._content

        def iter_lines(self):
            for line in self.text.splitlines():
                yield line

        def iter_bytes(self, chunk_size=None):
            yield self._content

        def raise_for_status(self):
            if self.status_code >= 400:
                raise RuntimeError(f"HTTP {self.status_code} for {self.url}")

        def close(self):
            pass

        def __enter__(self):
            return self

        def __exit__(self, *args):
            pass
    
    # Define patched functions
    def patched_get_with_retry(url, *args, **kwargs):
        content = curl_fetch(url, as_text=False)
        return FakeResponse(url, content, status_code=200)
    
    TEXT_EXTENSIONS = ('.htm', '.html', '.txt', '.sgml', '.xml', '.json', '.xsd')

    def patched_download_file(url, as_text=None, path=None):
        content = curl_fetch(url, as_text=False)
        if path:
            with open(path, "wb") as f:
                f.write(content)
            return path
        if as_text is False:
            return content
        if as_text or any(url.lower().split('?')[0].endswith(ext) for ext in TEXT_EXTENSIONS):
            return content.decode("utf-8", errors="replace")
        return content
    
    def patched_download_json(url):
        return json.loads(curl_fetch(url, as_text=True))
    
    def patched_stream_with_retry(url, *args, **kwargs):
        content = curl_fetch(url, as_text=False)
        yield FakeResponse(url, content, status_code=200)
    
    def patched_download_text(url, *args, **kwargs):
        content = curl_fetch(url, as_text=True)
        return content

    # Import edgar and patch all modules that use HTTP functions
    import edgar.httprequests as hr
    from edgar import set_identity

    # Patch the source module first
    hr.get_with_retry = patched_get_with_retry
    hr.download_file = patched_download_file
    hr.download_json = patched_download_json
    hr.download_text = patched_download_text
    hr.stream_with_retry = patched_stream_with_retry

    # Dynamically patch every edgar submodule that imported these functions
    import edgar
    import pkgutil
    patch_map = {
        "get_with_retry": patched_get_with_retry,
        "download_file": patched_download_file,
        "download_json": patched_download_json,
        "download_text": patched_download_text,
        "stream_with_retry": patched_stream_with_retry,
    }
    patched_count = 0
    for _imp, modname, _ispkg in pkgutil.walk_packages(edgar.__path__, prefix="edgar."):
        try:
            mod = __import__(modname, fromlist=[""])
            if mod is hr:
                continue
            for fn_name, fn_impl in patch_map.items():
                if hasattr(mod, fn_name):
                    setattr(mod, fn_name, fn_impl)
                    patched_count += 1
        except Exception:
            pass

    set_identity(UA)

    logger.info("EDGAR library patched with NTLM proxy (curl): %d sub-patches applied", patched_count)
    logger.info("   User-Agent: %s", UA)
